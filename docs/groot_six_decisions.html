env = gym.make("h1hand-window-v0", render_mode="rgb_array")

CHUNK_HORIZON = 40   # GR00T 输出
EXECUTE_FIRST = 10   # 只执行前 10 步
chunk = None; t_chunk = 0

with mujoco.viewer.launch_passive(base.model, base.data) as viewer:
    while viewer.is_running():
        if chunk is None or t_chunk >= EXECUTE_FIRST:
            obs_dict = build_groot_obs(env, obs)              # nested dict
            sock.send_pyobj({"obs": obs_dict})
            chunk = sock.recv_pyobj()["action"][0]            # (40, 28)
            t_chunk = 0
        a_groot = chunk[t_chunk]                              # (28,) arms+hands
        a_hb = compose_hb_action(arms_hands=a_groot,
                                 legs=STANDING_PRESET)        # → (61,)
        obs, _, term, trunc, _ = env.step(a_hb)
        viewer.sync()
        t_chunk += 1
        if term or trunc: obs, _ = env.reset(); chunk = None

Window FSM 设计（Codex 草案）

# humanoid_training/hb/window_controller.py
class WindowFSM:
    STATES = ["approach", "grip", "slide", "release"]

    def reset(self, env):
        self.state = "approach"; self.t = 0
        self.handle_pos = env.unwrapped.get_handle_pose()    # from htarget_high bbox
        self.target_slide_dist = 0.30                          # 30 cm 推开

    def target_action(self, obs):
        ee_pos = obs_to_ee_pos(obs, side="right")
        if self.state == "approach":
            arm_target = ik_solve(target=self.handle_pos + [0,0,0.05])
            hand = HAND_PRESETS["open"]
            if dist(ee_pos, self.handle_pos) < 0.03: self.state = "grip"
        elif self.state == "grip":
            arm_target = ik_solve(target=self.handle_pos)
            hand = HAND_PRESETS["half"]
            if self.t > 10: self.state = "slide"          # hold 10 ticks
        elif self.state == "slide":
            arm_target = ik_solve(target=self.handle_pos + [self.target_slide_dist, 0, 0])
            hand = HAND_PRESETS["half"]
            if dist_along_x > self.target_slide_dist - 0.02: self.state = "release"
        else:
            arm_target = ee_pos + [0, 0, 0.1]                # 抬起 10cm
            hand = HAND_PRESETS["open"]
        self.t += 1
        return np.concatenate([arm_target, hand,
                               np.zeros(14)])                  # left arm+hand zero

Success 判定：HB 内置 info["success"] + window joint progress > threshold（不要一上来搞双手 / 工具 / 移动 base）。

决策 6 · Eval & 训练监控

共识：N=10 ep × 3 seeds，watcher 重写不复用 slice early-stop

项方案
Eval 协议固定 N=10 episode × 3 seeds（与 DR.Q sister 路线对齐），用 HB 自带 info["success"] + success_bar
报告指标mean success rate · per-episode wall-clock · 失败阶段分布（approach / grip / slide / release 哪步失败）
训练监控 v1GR00T 自带 trainer_state.json loss 曲线 + 每 N checkpoint 跑 30-episode eval
Watcher 复用scripts/train_watcher.py 改造成 ckpt-eval watcher（每出 ckpt → mirror → eval → 写 JSONL）— 不沿用 SO-101 的 slice early-stop 逻辑（imitation 没有 RL 的 reward 曲线，slice 没意义）
Early-stop 策略看 ckpt-eval success rate 滑窗：连续 3 个 ckpt 不增长 → 早停
🅱：train_watcher.py 可以复用成 checkpoint/eval watcher，但别用 SO-101 的 slice early-stop 逻辑硬套。

6. 风险与退路 / Risk register

风险概率退路
WindowFSM IK 不收敛（H1 7D arm 解空间）中退到 joint-space 关键帧插值，不用 IK
50 ep 数据不够，finetune v1 success=0%中加 20 ep + state_dropout_prob 提高到 0.3 + image aug 加强

HB MuJoCo 渲染 OOD 致命中-高image aug + albumentations（sister repo 已开 use_albumentations_transforms: true）；最坏退到 IK pure scripted（不需要 GR00T）
